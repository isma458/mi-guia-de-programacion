body { margin: 0; font-family: Arial, sans-serif; background-color: #111; color: #fff; display: flex; justify-content: center; align-items: center; flex-direction: column; text-align: center; } button { margin: 5px; padding: 10px 20px; font-size: 16px; cursor: pointer; } .hidden { display: none; } #gameCanvas { border: 2px solid #fff; background: url('fondos/galaxia.jpg') no-repeat center; background-size: cover; margin-top: 20px; } // Variables del juego const menu = document.getElementById('menu'); const personajesDiv = document.getElementById('personajes'); const info = document.getElementById('info'); const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); let personaje = localStorage.getItem('personaje') || 'astro'; let nivel = parseInt(localStorage.getItem('nivel')) || 1; // Posición jugador let player = { x: 50, y: 300, width: 50, height: 50, dy: 0 }; let gravedad = 1; let saltando = false; // Obstáculos let obstaculos = []; let velocidad = 3; let puntaje = 0; // Mostrar personajes function mostrarPersonajes() { personajesDiv.classList.toggle('hidden'); } // Elegir personaje function elegirPersonaje(p) { personaje = p; localStorage.setItem('personaje', personaje); info.innerText = `Personaje elegido: ${personaje}`; } // Iniciar juego function iniciarJuego() { menu.classList.add('hidden'); canvas.classList.remove('hidden'); crearObstaculos(); requestAnimationFrame(gameLoop); } // Saltar document.addEventListener('keydown', (e) => { if(e.code === 'Space' && !saltando){ player.dy = -15; saltando = true; } }); // Crear obstáculos periódicamente function crearObstaculos() { setInterval(() => { obstaculos.push({ x: 800, y: 350, width: 30, height: 50 }); }, 2000); } // Game Loop function gameLoop() { ctx.clearRect(0, 0, canvas.width, canvas.height); // Dibujar jugador ctx.fillStyle = personaje === 'astro' ? 'yellow' : 'cyan'; ctx.fillRect(player.x, player.y, player.width, player.height); // Física jugador player.dy += gravedad; player.y += player.dy; if(player.y >= 300) { player.y = 300; player.dy = 0; saltando = false; } // Dibujar obstáculos y colisiones ctx.fillStyle = 'red'; obstaculos.forEach((obs, index) => { obs.x -= velocidad; ctx.fillRect(obs.x, obs.y, obs.width, obs.height); if(player.x < obs.x + obs.width && player.x + player.width > obs.x && player.y < obs.y + obs.height && player.y + player.height > obs.y){ // Colisión alert(`¡Game Over! Puntaje: ${puntaje}`); obstaculos = []; player.y = 300; puntaje = 0; menu.classList.remove('hidden'); canvas.classList.add('hidden'); return; } // Aumentar puntaje if(obs.x + obs.width < 0){ obstaculos.splice(index, 1); puntaje++; localStorage.setItem('nivel', nivel); // guardar progreso } }); // Mostrar puntaje ctx.fillStyle = 'white'; ctx.font = '20px Arial'; ctx.fillText(`Nivel: ${nivel} Puntaje: ${puntaje}`, 10, 30); requestAnimationFrame(gameLoop); }
